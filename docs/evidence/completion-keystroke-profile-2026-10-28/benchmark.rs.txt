use std::time::Instant;

use tablepro_app_services::completion::SchemaIndex;
use tablepro_app_services::completion_proposals::candidates;
use tablepro_core::sql_syntax::SqlGrammar;

fn mean_ms(warmup: u32, reps: u32, mut run: impl FnMut() -> usize) -> (f64, usize) {
    let mut matches = 0;
    for _ in 0..warmup {
        matches = std::hint::black_box(run());
    }
    let started = Instant::now();
    for _ in 0..reps {
        matches = std::hint::black_box(run());
    }
    (started.elapsed().as_secs_f64() * 1e3 / reps as f64, matches)
}

fn index_with(tables: usize) -> SchemaIndex {
    let mut index = SchemaIndex::default();
    index.grammar = Some(SqlGrammar::PostgreSql);
    index.set_tables((0..tables).map(|i| format!("tenant_{:03}.table_{i:06}", i % 200)).collect());
    index
}

fn script(statements: usize) -> String {
    let mut sql = String::new();
    for i in 0..statements {
        sql.push_str(&format!(
            "-- row {i}\nINSERT INTO t{} (id, note, body) VALUES ({i}, 'it''s {i}; not a split', $$dollar; quoted {i}$$);\n/* block; comment {i} */\nSELECT a, b FROM \"t;{}\" WHERE x = {i};\n",
            i % 50,
            i % 7
        ));
    }
    sql
}

fn main() {
    let mut rows = Vec::new();
    for tables in [1_000usize, 10_000, 100_000] {
        let index = index_with(tables);
        for (label, sql) in [("selective", "SELECT * FROM tenant_007.table_0000"), ("no_match", "SELECT * FROM zzz")] {
            let (ms, matches) = mean_ms(2, 20, || candidates(sql, sql.len(), &index).values.len());
            rows.push(format!(
                r#"{{"axis":"catalog","tables":{tables},"prefix":"{label}","matches":{matches},"ms":{ms:.4}}}"#
            ));
        }
    }
    for statements in [100usize, 2_000, 20_000] {
        let mut sql = script(statements);
        sql.push_str("SELECT * FROM tenant_007.table_0000");
        let index = index_with(50);
        let bytes = sql.len();
        let (ms, matches) = mean_ms(1, 5, || candidates(&sql, sql.len(), &index).values.len());
        rows.push(format!(r#"{{"axis":"script","script_bytes":{bytes},"tables":50,"matches":{matches},"ms":{ms:.4}}}"#));
    }
    println!("[{}]", rows.join(",\n "));
}
